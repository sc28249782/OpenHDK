cmake_minimum_required(VERSION 3.22)
project(OpenHDK VERSION 0.1.0 LANGUAGES C CXX)
option(OPENHDK_BUILD_TESTS "Build OpenHDK tests" ON)
option(OPENHDK_FETCH_TEST_FIXTURES "Download pinned test fixtures" ON)
include(FetchContent)
FetchContent_Declare(miniaudio GIT_REPOSITORY https://github.com/mackron/miniaudio.git GIT_TAG 9634bedb5b5a2ca38c1ee7108a9358a4e233f14d GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(miniaudio)
find_package(FluidSynth CONFIG REQUIRED)
add_library(openhdk_miniaudio STATIC "${miniaudio_SOURCE_DIR}/miniaudio.c")
add_library(OpenHDK::miniaudio ALIAS openhdk_miniaudio)
target_include_directories(openhdk_miniaudio SYSTEM PUBLIC "${miniaudio_SOURCE_DIR}")
set_target_properties(openhdk_miniaudio PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
add_library(openhdk_core STATIC audio/FluidSynthBackend.cpp)
add_library(OpenHDK::core ALIAS openhdk_core)
target_include_directories(openhdk_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(openhdk_core PUBLIC cxx_std_20)
target_link_libraries(openhdk_core PRIVATE FluidSynth::libfluidsynth OpenHDK::miniaudio)
add_executable(OpenHDK app/main.cpp)
target_link_libraries(OpenHDK PRIVATE OpenHDK::core)
if(OPENHDK_BUILD_TESTS)
  include(CTest)
  if(BUILD_TESTING)
    if(NOT OPENHDK_FETCH_TEST_FIXTURES)
      message(FATAL_ERROR "Tests require OPENHDK_FETCH_TEST_FIXTURES=ON.")
    endif()
    set(OPENHDK_TEST_FIXTURE_DIR "${CMAKE_BINARY_DIR}/test-fixtures")
    set(OPENHDK_TEST_SOUNDFONT "${OPENHDK_TEST_FIXTURE_DIR}/VintageDreamsWaves-v2.sf2")
    file(MAKE_DIRECTORY "${OPENHDK_TEST_FIXTURE_DIR}")
    file(DOWNLOAD "https://raw.githubusercontent.com/FluidSynth/fluidsynth/3ede3f7c8ffc5d16e5d28f416cbfd71f3ad2b814/sf2/VintageDreamsWaves-v2.sf2" "${OPENHDK_TEST_SOUNDFONT}" EXPECTED_HASH "SHA256=52132b2b83994f0067d3a66b93b4f1b67d53ff8c0d3be90d39f23f92b4585cdd" TLS_VERIFY ON STATUS OPENHDK_TEST_SOUNDFONT_STATUS)
    list(GET OPENHDK_TEST_SOUNDFONT_STATUS 0 OPENHDK_TEST_SOUNDFONT_RESULT)
    if(NOT OPENHDK_TEST_SOUNDFONT_RESULT EQUAL 0)
      list(GET OPENHDK_TEST_SOUNDFONT_STATUS 1 OPENHDK_TEST_SOUNDFONT_ERROR)
      message(FATAL_ERROR "Could not acquire pinned SoundFont: ${OPENHDK_TEST_SOUNDFONT_ERROR}")
    endif()
    add_executable(openhdk_bootstrap_smoke tests/bootstrap_smoke.cpp)
    target_link_libraries(openhdk_bootstrap_smoke PRIVATE OpenHDK::core)
    target_compile_definitions(openhdk_bootstrap_smoke PRIVATE OPENHDK_TEST_SOUNDFONT_PATH="${OPENHDK_TEST_SOUNDFONT}")
    add_test(NAME fluidsynth-headless-poc COMMAND openhdk_bootstrap_smoke)
  endif()
endif()
